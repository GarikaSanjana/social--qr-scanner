<script>
const results = document.getElementById('results');
function handleDecodedText(text) {
  results.innerHTML = '';
  try {
    const j = JSON.parse(text);
    if (j && j.type === 'social_list' && j.accounts) {
      for (const [n,u] of Object.entries(j.accounts)) {
        const div = document.createElement('div');
        div.innerHTML = `<a href="${u}" target="_blank">${n}</a>`;
        results.appendChild(div);
      }
      return;
    }
  } catch(e){}
  if (/^https?:/.test(text)) {
    results.innerHTML = `<p>Scanned URL: <a href="${text}" target="_blank">${text}</a></p>`;
  } else {
    results.textContent = text;
  }
}
const html5QrCode = new Html5Qrcode("reader");
Html5Qrcode.getCameras().then(cameras => {
  const camId = cameras && cameras.length ? cameras[0].id : null;
  if (camId) {
    html5QrCode.start(camId, { fps: 10, qrbox: 250 },
      (decodedText) => {
        html5QrCode.stop().then(()=>handleDecodedText(decodedText));
      });
  } else {
    results.textContent = 'No camera found.';
  }
});
</script>
