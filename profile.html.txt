<script>
const accounts = {
  "Instagram": "https://instagram.com/yourusername",
  "LinkedIn": "https://linkedin.com/in/yourusername",
  "GitHub": "https://github.com/yourusername",
  "Twitter": "https://twitter.com/yourusername",
  "YouTube": "https://youtube.com/@yourusername"
};

const linksList = document.getElementById('linksList');
for (const [name, url] of Object.entries(accounts)) {
  const a = document.createElement('a');
  a.href = url; a.target = '_blank'; a.textContent = `🔗 ${name}`;
  linksList.appendChild(a);
}

const qrcodeEl = document.getElementById('qrcode');
new QRCode(qrcodeEl, { text: location.href, width: 220, height: 220, correctLevel: QRCode.CorrectLevel.H });

document.getElementById('downloadBtn').addEventListener('click', () => {
  const img = qrcodeEl.querySelector('img');
  const a = document.createElement('a');
  a.href = img.src; a.download = 'my_socials_qr.png'; a.click();
});
</script>
